model DomainItem {
  // Primary key
  id          String   @id @default(cuid())
  
  // Multi-tenancy (REQUIRED)
  school_id   String
  
  // Core fields
  name        String
  description String?  // Optional
  quantity    Int      @default(0)
  status      DomainStatus @default(ACTIVE)
  
  // Metadata
  metadata    Json?    @default("{}")
  
  // Timestamps
  created_at  DateTime @default(now()) @db.Timestamptz(6)
  updated_at  DateTime @updatedAt
  
  // Relations
  school      School   @relation(fields: [school_id], references: [id])
  created_by  User?    @relation("DomainItemCreator", fields: [created_by_id], references: [id])
  created_by_id String?
  
  // Indexes (REQUIRED for multi-tenancy performance)
  @@index([school_id])
  @@index([school_id, status])
  
  // Schema tag (REQUIRED)
  @@schema("schoolmanagementsystem")
}

// ----------------------------------------------------------------------------
// ENUM TEMPLATE
// ----------------------------------------------------------------------------

enum DomainStatus {
  ACTIVE
  INACTIVE
  ARCHIVED
  
  @@schema("schoolmanagementsystem")
}

// ----------------------------------------------------------------------------
// RELATION EXAMPLES
// ----------------------------------------------------------------------------

// One-to-Many: School has many Items
// In School model, add:
//   domainItems  DomainItem[]

// One-to-One: User has one Profile
model Profile {
  id      String @id @default(cuid())
  user_id String @unique  // Note: @unique for one-to-one
  user    User   @relation(fields: [user_id], references: [id])
  
  @@schema("schoolmanagementsystem")
}

// Many-to-Many: Students enrolled in Subjects
model StudentSubject {
  student_id String
  subject_id String
  
  student    Student @relation(fields: [student_id], references: [id])
  subject    Subject @relation(fields: [subject_id], references: [id])
  
  // Additional fields on the join table
  enrolled_at DateTime @default(now())
  
  @@id([student_id, subject_id])
  @@schema("schoolmanagementsystem")
}
